cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0141 NEW)
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT Embedded)
project(ComponentAccessProbe LANGUAGES CXX)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 20)
add_library(ComponentAccessProbe SHARED Native.cpp)
foreach(dir Funke-Engine/src Funke-Engine/src/Generated External/spdlog/include External/glm External/entt/src External/stb External/magic_enum/include External/rapidxml External/concurrentqueue External/SDL/include External/imgui External/dawn/include External/dawn/build/gen/include External/miniaudio External/box2d/include External/JoltPhysics External/dotnet External/tracy/public)
  target_include_directories(ComponentAccessProbe PRIVATE "${REPO}/${dir}")
endforeach()
target_compile_definitions(ComponentAccessProbe PRIVATE UNICODE _UNICODE FNK_IMPORT_DLL FNK_PLATFORM_WINDOWS FNK_RELEASE NDEBUG NOMINMAX SPDLOG_WCHAR_FILENAMES FNK_RHI_WEBGPU FUNKE_ALL_MODULES=1 FUNKE_WITH_EDITOR=1 FUNKE_WITH_2D=1 FUNKE_WITH_3D=1 FUNKE_WITH_PHYSICS2D=1 FUNKE_WITH_PHYSICS3D=1 FUNKE_ENTITY_BITS=20 FUNKE_PROFILER_ENABLED TRACY_ENABLE TRACY_IMPORTS TRACY_ON_DEMAND)
target_compile_options(ComponentAccessProbe PRIVATE /utf-8 /O2 /wd4251 /wd4275)
target_link_libraries(ComponentAccessProbe PRIVATE "${REPO}/bin/Release-windows-x86_64/Funke-Engine/Funke-Engine.lib")
