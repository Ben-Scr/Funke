cmake_minimum_required(VERSION 3.22)
project(FunkeWebEditor C CXX)
option(FUNKE_WEB_EDITOR_MANAGED "Link the editor into the .NET browser runtime" OFF)
if(NOT EMSCRIPTEN)
    message(FATAL_ERROR "This experimental editor target requires Emscripten.")
endif()
foreach(required FUNKE_ROOT FUNKE_SDL3_DIR FUNKE_EMDAWNWEBGPU_PORT FUNKE_WEBGPU_INCLUDE_DIRS FUNKE_EDITOR_CONTENT)
    if(NOT DEFINED ${required})
        message(FATAL_ERROR "${required} is required; use Editor/build.py")
    endif()
endforeach()
add_compile_options(-fwasm-exceptions -msimd128 -msse4.2)
add_link_options(-fwasm-exceptions -msimd128)
if(FUNKE_WEB_EDITOR_MANAGED)
    add_compile_options(-sWASM_LEGACY_EXCEPTIONS=0)
    add_link_options(-sWASM_LEGACY_EXCEPTIONS=0)
endif()
set(_port "--use-port=${FUNKE_EMDAWNWEBGPU_PORT}")
add_compile_options("${_port}")
include("${FUNKE_ROOT}/cmake/FunkeEngine.cmake")
add_library(SDL3::SDL3 STATIC IMPORTED)
set_target_properties(SDL3::SDL3 PROPERTIES IMPORTED_LOCATION "${FUNKE_SDL3_DIR}/lib/libSDL3.a" INTERFACE_INCLUDE_DIRECTORIES "${FUNKE_SDL3_DIR}/include")
add_library(FunkeWebGPU INTERFACE)
target_link_options(FunkeWebGPU INTERFACE "${_port}")
funke_add_engine(CONFIGURATION Development ENTITY_BITS 20 SDL3_TARGET SDL3::SDL3 WEBGPU_TARGET FunkeWebGPU WEBGPU_INCLUDE_DIRS ${FUNKE_WEBGPU_INCLUDE_DIRS} MODULES Funke.2D Funke.3D Funke.Physics2D)
# Player exports strip editor APIs. This target retains them without altering player builds.
foreach(_property COMPILE_DEFINITIONS INTERFACE_COMPILE_DEFINITIONS)
    get_target_property(_defines FunkeEngine ${_property})
    list(REMOVE_ITEM _defines FUNKE_WITH_EDITOR=0)
    list(APPEND _defines FUNKE_WITH_EDITOR=1 FUNKE_WEB_EDITOR=1)
    set_property(TARGET FunkeEngine PROPERTY ${_property} "${_defines}")
endforeach()
file(GLOB_RECURSE _editor_sources CONFIGURE_DEPENDS "${FUNKE_ROOT}/Funke-Editor/src/*.cpp" "${FUNKE_ROOT}/Funke-EditorRuntime/src/*.cpp")
list(FILTER _editor_sources EXCLUDE REGEX "/(EditorApplication|EditorPch)\\.cpp$|/Bridge/")
file(GLOB _diagnostics CONFIGURE_DEPENDS "${FUNKE_ROOT}/Funke-Engine/src/Diagnostics/*Overlay.cpp")
file(GLOB _gizmo CONFIGURE_DEPENDS "${FUNKE_EXTERNAL}/ImGuizmo/src/*.cpp")
add_library(FunkeWebEditorObjects OBJECT ${_editor_sources} ${_diagnostics})
target_include_directories(FunkeWebEditorObjects PUBLIC "${FUNKE_ROOT}/Funke-Editor/src" "${FUNKE_ROOT}/Funke-EditorRuntime/src" "${FUNKE_EXTERNAL}/ImGuizmo/src" "${FUNKE_EXTERNAL}/md4c")
target_link_libraries(FunkeWebEditorObjects PUBLIC FunkeEngine)
target_precompile_headers(FunkeWebEditorObjects PRIVATE "${FUNKE_ROOT}/Funke-Editor/src/EditorPch.hpp")
_funke_apply_common_options(FunkeWebEditorObjects)
add_library(FunkeWebEditorExtras STATIC ${_gizmo} "${FUNKE_EXTERNAL}/md4c/md4c.c")
target_link_libraries(FunkeWebEditorExtras PUBLIC FunkeImGui)
if(FUNKE_WEB_EDITOR_MANAGED)
    add_library(FunkeWeb STATIC WebEditorMain.cpp "${FUNKE_MONO_PACKAGE_DIR}/Native/MonoScriptRuntime.cpp")
    target_include_directories(FunkeWeb PRIVATE "${FUNKE_MONO_INCLUDE_DIR}" "${FUNKE_MONO_PACKAGE_DIR}/Native")
    target_compile_definitions(FunkeWeb PRIVATE FUNKE_WEB_EDITOR_MANAGED=1 FUNKE_CUSTOM_MAIN)
    target_link_libraries(FunkeWeb PRIVATE FunkeWebEditorObjects FunkeWebEditorExtras FunkeEngine)
    set_target_properties(FunkeWeb PROPERTIES PREFIX "")
    _funke_apply_common_options(FunkeWeb)
    return()
endif()
add_executable(index WebEditorMain.cpp)
target_link_libraries(index PRIVATE FunkeWebEditorObjects FunkeWebEditorExtras FunkeEngine)
target_compile_definitions(index PRIVATE FUNKE_CUSTOM_MAIN)
_funke_apply_common_options(index)
set_target_properties(index PROPERTIES SUFFIX ".html")
set_property(TARGET index APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/shell.html")
target_link_options(index PRIVATE "--shell-file=${CMAKE_CURRENT_LIST_DIR}/shell.html" "--preload-file=${FUNKE_EDITOR_CONTENT}@/editor" -sALLOW_MEMORY_GROWTH=1 -sINITIAL_MEMORY=268435456 -sSTACK_SIZE=8388608 -sEXIT_RUNTIME=0 -sFORCE_FILESYSTEM=1 -sENVIRONMENT=web -lidbfs.js "-sEXPORTED_RUNTIME_METHODS=['FS','ccall','UTF8ToString']")
